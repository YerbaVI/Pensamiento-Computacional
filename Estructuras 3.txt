//Estructuras 3
#include <iostream>
using namespace std;
struct texto{
    string nombre;
};
int main(){
    int n;
    cout << "Ingrese la cantidad de textos: ";
    cin >>n;
    texto A[n];
    for (int i=0;i<n;i++){
        cout << "Texto" << i+1 << " : ";
        cin >> A[i].nombre;
    }
    cout << "-- los textos son --" << endl;
    for (int i=0;i<n;i++){
        cout << A[i].nombre << endl;
    }
    return 0;
}
