#include <iostream>
using namespace std;
int factorial(int n){
    if (n<=1){
        return 1;
    }else{
        return n * factorial(n-1);
    }
}
int main (){
    int A;
    cout << "Ingrese la variable: ";
    cin >> A;
    cout << "El factorial es: " << factorial(A);
    return 0;
}
